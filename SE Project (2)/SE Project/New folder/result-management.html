function getGrade(pct) { var s = JSON.parse(localStorage.getItem('sysSettings') || '{}'); var aPlus = s.gradeAPlus !== undefined ? parseFloat(s.gradeAPlus) : 90; var a = s.gradeA !== undefined ? parseFloat(s.gradeA) : 80; var b = s.gradeB !== undefined ? parseFloat(s.gradeB) : 70; var c = s.gradeC !== undefined ? parseFloat(s.gradeC) : 60; var d = s.gradeD !== undefined ? parseFloat(s.gradeD) : 50; if (pct >= aPlus) return 'A+'; if (pct >= a) return 'A'; if (pct >= b) return 'B'; if (pct >= c) return 'C'; if (pct >= d) return 'D'; return 'F'; } function populateExamSelect() { var sel = document.getElementById('selectExam'); var exams = getExams(); sel.innerHTML = ''; exams.forEach(function(e) { sel.innerHTML += ''; }); } function loadExamStudents() { var examId = parseInt(document.getElementById('selectExam').value); var cls = document.getElementById('filterClass').value; var sec = document.getElementById('filterSection').value; var tbody = document.getElementById('marksBody'); if (!examId) { tbody.innerHTML = 'Select an exam to enter marks.'; return; } var exam = getExams().find(function(e){ return e.id===examId; }); if (!exam) return; document.getElementById('examInfo').textContent = exam.title + ' | Total: ' + exam.marks + ' | Pass: ' + exam.passing; var students = getStudents().filter(function(s) { var mc = !cls || s.class === cls; var ms = !sec || s.section === sec; var me = exam.class === 'All Classes' || s.class === exam.class; return mc && ms && me; }); var savedResults = getResults(); if (students.length === 0) { tbody.innerHTML = 'No students found for this exam/filter.'; return; } tbody.innerHTML = students.map(function(s, i) { var existing = savedResults.find(function(r){ return r.studentId===s.id && r.examId===examId; }); var obtained = existing ? existing.obtained : ''; var pct = obtained !== '' ? Math.round((obtained / exam.marks) * 100) : ''; var grade = pct !== '' ? getGrade(pct) : ''; var pass = pct !== '' ? (obtained >= exam.passing) : null; return '' + ''+(i+1)+'' + ''+(s.rollNo||'-')+'' + ''+s.name+'' + ''+s.class+'' + ''+(s.section||'-')+'' + ''+exam.marks+'' + '' + ''+(pct!==''?pct+'%':'-')+'' + ''+(grade?''+grade+'':'-')+'' + ''+(pass===null?'-':pass?'Pass':'Fail')+'' + ''; }).join(''); } function calcRow(studentId, totalMarks, passingMarks) { var val = parseFloat(document.getElementById('marks-'+studentId).value); if (isNaN(val) || val < 0) { document.getElementById('pct-'+studentId).textContent='-'; document.getElementById('grade-'+studentId).textContent='-'; document.getElementById('result-'+studentId).textContent='-'; return; } var pct = Math.round((val / totalMarks) * 100); var grade = getGrade(pct); var pass = val >= passingMarks; document.getElementById('pct-'+studentId).textContent = pct + '%'; document.getElementById('grade-'+studentId).innerHTML = ''+grade+''; document.getElementById('result-'+studentId).innerHTML = pass ? 'Pass' : 'Fail'; } function saveAllResults() { var examId = parseInt(document.getElementById('selectExam').value); if (!examId) { alert('Please select an exam first.'); return; } var exam = getExams().find(function(e){ return e.id===examId; }); var inputs = document.querySelectorAll('[id^="marks-"]'); if (inputs.length === 0) { alert('No students loaded.'); return; } var results = getResults().filter(function(r){ return r.examId !== examId; }); var count = 0; inputs.forEach(function(inp) { var studentId = parseInt(inp.id.replace('marks-','')); var val = parseFloat(inp.value); if (isNaN(val)) return; var pct = Math.round((val / exam.marks) * 100); var student = getStudents().find(function(s){ return s.id===studentId; }); results.push({ id: Date.now() + Math.random(), examId: examId, examTitle: exam.title, subject: exam.subject, studentId: studentId, studentName: student ? student.name : '-', rollNo: student ? student.rollNo : '-', totalMarks: exam.marks, obtained: val, percentage: pct, grade: getGrade(pct), pass: val >= exam.passing }); count++; }); saveResults(results); updateStats(); renderSavedResults(); alert(count + ' results saved successfully.'); } function renderSavedResults() { var results = getResults(); var tbody = document.getElementById('resultsBody'); if (results.length === 0) { tbody.innerHTML = 'No results saved yet.'; return; } tbody.innerHTML = results.slice(0,50).map(function(r, i) { return ''+(i+1)+''+r.studentName+'
